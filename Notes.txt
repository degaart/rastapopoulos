# Build grub on osx
../configure --prefix=/opt/grub --disable-dependency-tracking --disable-nls --disable-werror  --target=i686-elf

# create an image on osx

# Install grub on osx
/opt/grub/sbin/grub-install --target=i386-pc --modules="part_msdos biosdisk fat multiboot configfile" --root-directory=/Volumes/

# /boot/grub/grub.cfg
set timeout=15
set default=0 # Set the default menu entry
 
menuentry "OS Name" {
   multiboot /boot/kernel-file   # The multiboot command replaces the kernel command
   boot
}
